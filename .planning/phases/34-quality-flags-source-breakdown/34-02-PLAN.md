    3. Create a local reactive `qualityExclusionConfig` ref initialized from store defaults (per Pitfall 7 in RESEARCH.md -- per-analysis local state, not persisted back to store on every toggle):
       ```typescript

         ...qualityStore.exclusionDefaults,
       });
       ```
       Also create a setter:
       ```typescript
       const setQualityExclusionConfig = (config: QualityExclusionConfig) => {
         qualityExclusionConfig.value = { ...config };
       };
       ```

    4. Create a computed that computes quality flags for all filtered variants (variants that passed pathogenicity filtering but BEFORE manual/quality exclusions). This is computed from the existing `filterPathogenicVariantsConfigurable` result. Add a new computed BEFORE the existing `pathogenicVariants` computed:

       ```typescript
       // Variants that pass pathogenicity filters (before any exclusions)
       const filteredByPathogenicity = computed(() => {
         if (!normalizedVariants.value.length) return [];
         return filterPathogenicVariantsConfigurable(
           normalizedVariants.value,
           normalizedClinvar.value,
           filterConfig.value,
           submissions.value,
         );
       });
       ```

       Then create a quality flags map computed:
       ```typescript
       const qualityFlagsMap = computed((): Map<string, QualityFlag[]> => {
         const map = new Map();
         for (const variant of filteredByPathogenicity.value) {
           const flags = computeQualityFlags(variant, qualityStore.defaults);
           if (flags.length > 0) {
             map.set(variant.variant_id, flags);
           }
         }
         return map;
       });
       ```

    5. Create quality exclusion computed:
       ```typescript
       const qualityExcludedIds = computed((): Set<string> => {
         const excluded = new Set<string>();
         for (const [variantId, flags] of qualityFlagsMap.value) {
           if (shouldExcludeByQuality(flags, qualityExclusionConfig.value)) {
             excluded.add(variantId);
           }
         }
         return excluded;
       });

       const qualityExcludedCount = computed(() => qualityExcludedIds.value.size);
       ```

    6. Update `totalPathogenicCount` to use `filteredByPathogenicity`:
       ```typescript
       const totalPathogenicCount = computed(() => filteredByPathogenicity.value.length);
       ```

    7. Update `pathogenicVariants` to filter out BOTH manual AND quality exclusions:
       ```typescript
       const pathogenicVariants = computed(() => {
         return filteredByPathogenicity.value.filter(v =>
           !debouncedExcluded.value.has(v.variant_id) &&
           !qualityExcludedIds.value.has(v.variant_id)
         );
       });
       ```

    8. Add a `flaggedVariantCount` computed for QUAL-06 (summary count of flagged variants):
       ```typescript
       const flaggedVariantCount = computed(() => qualityFlagsMap.value.size);
       ```

    9. Update the `UseCarrierFrequencyReturn` interface to add new fields:
