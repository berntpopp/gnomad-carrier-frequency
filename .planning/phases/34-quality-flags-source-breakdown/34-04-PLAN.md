       ```typescript
       const sourceCategoryMap = computed((): Map<string, SourceCategory> => {
         const map = new Map<string, SourceCategory>();
         // filteredByPathogenicity contains variants after pathogenicity filtering
         // but before manual/quality exclusions -- exactly the set we want to classify
         for (const variant of filteredByPathogenicity.value) {

             classifyVariantSource(variant, clinvarVariants.value, filterConfig.value, submissions.value),
           );
         }
         return map;
       });

       function getVariantFlags(variantId: string): QualityFlag[] {

       function worstFlagColor(flags: QualityFlag[]): string {
         // Return the most severe color: critical > warning > info
         if (flags.some(f => f.severity === 'critical')) return 'error';
         if (flags.some(f => f.severity === 'warning')) return 'warning';
         return 'blue-grey';
       }

       // For inline CSS color in tooltip text (Vuetify color names don't work in plain HTML)
       function flagTypeColorCss(type: QualityFlagType): string {
         const map: Record<QualityFlagType, string> = {

           high_hom: '#FF9800',      // orange
           gnomad_filtered: '#FFC107', // amber
           genomes_only: '#607D8B',    // blue-grey
         };
         return map[type];
       }
       ```
