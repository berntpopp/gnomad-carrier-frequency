    ```typescript
    import type { GnomadVariant, ClinVarVariant, FilterConfig, CalcConfig } from '../types/index.js';
    import type { SourceCategory } from '../filters/source-classification.js';
    import { classifyVariantSource } from '../filters/source-classification.js';
    import type { ClinVarSubmission } from '../queries/index.js';

    export interface SourceBreakdownRow {
      sourceCategory: SourceCategory;
      label: string;         // "ClinVar", "pLoF", "Both"
      variantCount: number;
      carrierFrequency: number | null;
      alleleCount: number;
      alleleNumber: number;
    }

    /**
     * Compute source breakdown for a given population.
     * Groups pathogenic variants by source category, then computes carrier frequency per group.
     *
     * @param variants - Pathogenic variants (already filtered, before manual/quality exclusions or after -- caller decides)
     * @param clinvarVariants - ClinVar variants for source classification
     * @param filterConfig - Current filter config (for source classification thresholds)
     * @param populationCode - Population to compute for
     * @param calcConfig - Calculation config (for HWE vs simplified formula)
     * @param submissionsMap - Optional conflicting submissions map (MUST be passed for correct conflicting-pathway classification)
     * @returns Array of SourceBreakdownRow, one per source category that has variants
     */
    export function computeSourceBreakdown(
      variants: GnomadVariant[],
      clinvarVariants: ClinVarVariant[],
      filterConfig: FilterConfig,
      populationCode: string,
      calcConfig: CalcConfig,
      submissionsMap?: Map<string, ClinVarSubmission[]>,
    ): SourceBreakdownRow[] {
      // Group variants by source category
      const groups: Record<SourceCategory, GnomadVariant[]> = {
        clinvar_only: [],
        plof_only: [],
        both: [],
      };

      for (const variant of variants) {
        const cat = classifyVariantSource(variant, clinvarVariants, filterConfig, submissionsMap);
        groups[cat].push(variant);
      }

      const rows: SourceBreakdownRow[] = [];
      const labels: Record<SourceCategory, string> = {
        clinvar_only: 'ClinVar',
        plof_only: 'pLoF',
        both: 'Both',
      };

      for (const cat of ['clinvar_only', 'plof_only', 'both'] as SourceCategory[]) {
        const groupVariants = groups[cat];
        if (groupVariants.length === 0) continue;

        // Compute allele frequency sum for this group in this population
        let sumAF = 0;
        let totalAC = 0;
        let maxAN = 0;

        for (const variant of groupVariants) {
          let ac: number;
          let an: number;

          // Get population-specific counts -- prefer joint data
          const jointPop = variant.joint?.populations?.find(p => p.id === populationCode);
          if (jointPop) {
            ac = jointPop.ac;
            an = jointPop.an;
          } else {
            const exomePop = variant.exome?.populations?.find(p => p.id === populationCode);
            const genomePop = variant.genome?.populations?.find(p => p.id === populationCode);
            ac = (exomePop?.ac ?? 0) + (genomePop?.ac ?? 0);
            an = (exomePop?.an ?? 0) + (genomePop?.an ?? 0);
          }

          totalAC += ac;
          maxAN = Math.max(maxAN, an);
          if (an > 0) {
            sumAF += ac / an;
          }
        }

        // Compute carrier frequency using same formula as global calculation
        let carrierFrequency: number | null = null;
        if (sumAF > 0) {
          if (calcConfig.useHWEFormula) {
            // 2pq where p = 1-q, q = sumAF
            const q = sumAF;
            carrierFrequency = 2 * (1 - q) * q;
          } else {
            carrierFrequency = 2 * sumAF;
          }
        }

        rows.push({
          sourceCategory: cat,
          label: labels[cat],
          variantCount: groupVariants.length,
          carrierFrequency,
          alleleCount: totalAC,
          alleleNumber: maxAN,
        });
      }

      return rows;
    }
    ```

    Update `packages/core/src/calculations/index.ts`:

    Note: This function is intentionally simplified. It does NOT use VCR/GCR (homozygote exclusion) for per-source breakdown because the per-source variant count may be too small for meaningful homozygote exclusion. It uses the HWE (2pq) or simplified (2*sumAF) formula based on CalcConfig.
