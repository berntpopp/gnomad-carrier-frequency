**Step 8: Echo equivalent command (key CONTEXT.md feature)**
```typescript
function buildEquivalentCommand(opts: {
  gene: string
  version: string
  populations: string[]
  format: string
  hwe: boolean
  excludeHomozygotes: boolean
  penetrance: number
  variants: boolean
}): string {
  const parts = ['gnomad-cf', 'query', opts.gene]
  if (opts.version !== 'v4') parts.push(`--gnomad-version ${opts.version}`)
  if (opts.populations.length > 0) {
    for (const pop of opts.populations) parts.push(`--population ${pop}`)
  }
  // Wait -- query command only accepts one --population. If multiple selected, this needs handling.
  // For now: if multiple populations selected, don't add --population (show all, user can filter in post-processing)
  // If exactly one selected, add --population
  if (opts.populations.length === 1) parts.push(`--population ${opts.populations[0]}`)
  if (opts.format !== 'text') parts.push(`--format ${opts.format}`)
  if (!opts.hwe) parts.push('--no-hwe')
  if (!opts.excludeHomozygotes) parts.push('--no-exclude-homozygotes')
  if (opts.penetrance !== 1.0) parts.push(`--penetrance ${opts.penetrance}`)
  if (opts.variants) parts.push('--variants')
  return parts.join(' ')
}

const cmd = buildEquivalentCommand({ gene, version, populations, format, hwe, excludeHomozygotes, penetrance, variants })
p.note(cmd, 'Equivalent command:')
```
