```typescript
function escapeTsv(value: string | number | boolean | null | undefined): string {
  if (value === null || value === undefined) return '""';
  const str = String(value).replace(/\n/g, " ").replace(/\t/g, " ").replace(/"/g, '""');
  return `"${str}"`;
}
```

Add two new exported functions:

1. **`buildPopulationsTsv(data: ExportData): string`**
   - Header: `Population\tCarrier Frequency\tRatio\tRecurrence Risk\tAC\tAN\tNotes`
   - For each population in `data.populations`:
     - Population: `pop.label`
     - Carrier Frequency: `pop.carrierFrequency` (raw decimal, NOT the percent string)
     - Ratio: `pop.carrierFrequencyRatio` (the "1:N" string)
     - Recurrence Risk: compute from carrier frequency. If `pop.carrierFrequency` is not null, calculate `pop.carrierFrequency / 4` (assume carrier index status for export, which is the common case). Format as raw decimal.
     - AC: `pop.alleleCount`
     - AN: `pop.alleleNumber`
     - Notes: `pop.isFounderEffect ? "Founder effect" : ""`
   - All fields escaped with `escapeTsv()`
   - Rows joined with `\n`, header as first line
   - Return the TSV string (NO BOM prefix -- the composable adds it)

2. **`buildVariantsTsv(data: ExportData): string`**
   - Header: `Variant ID\tConsequence\tAF\tCarrier Frequency\tClinVar Significance\tStars\tHGVS-c\tHGVS-p\tSource Category\tQuality Flags`
   - For each variant in `data.variants`:
     - Variant ID: `v.variantId`
     - Consequence: `v.consequence`
     - AF: `v.alleleFrequency` (raw decimal)
     - Carrier Frequency: if `v.alleleFrequency` is not null, `v.alleleFrequency * 2` (approximate carrier freq for single variant). Raw decimal.
     - ClinVar Significance: `v.clinvarStatus ?? ""`
     - Stars: `""` (Phase 34 will add gold stars to ExportVariant)
     - HGVS-c: `v.hgvsC ?? ""`
     - HGVS-p: `v.hgvsP ?? ""`
     - Source Category: `""` (Phase 34 placeholder)
     - Quality Flags: `""` (Phase 34 placeholder)
   - All fields escaped, rows joined with `\n`
   - Return TSV string (no BOM)

```typescript
function exportPopulationsTsv(data: ExportData, gene: string): void {
  const BOM = "\uFEFF";
  const tsv = BOM + buildPopulationsTsv(data);
  const blob = new Blob([tsv], { type: "text/tab-separated-values;charset=utf-8" });
  const date = new Date().toISOString().split("T")[0];
  const filename = `${sanitizeFilename(gene)}_populations_${date}.tsv`;
  downloadBlob(blob, filename);
}

function exportVariantsTsv(data: ExportData, gene: string): void {
  const BOM = "\uFEFF";
  const tsv = BOM + buildVariantsTsv(data);
  const blob = new Blob([tsv], { type: "text/tab-separated-values;charset=utf-8" });
  const date = new Date().toISOString().split("T")[0];
  const filename = `${sanitizeFilename(gene)}_variants_${date}.tsv`;
  downloadBlob(blob, filename);
}
```
